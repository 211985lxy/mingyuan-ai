```prisma
// ─── Competitor Analysis (v5.0) ───────────────────────────

model CompetitorAnalysis {
  id                String   @id @default(cuid())
  userId            String
  user              User     @relation(fields: [userId], references: [id])

  // Input
  targetUrl         String   @db.VarChar(500)
  platform          String   @db.VarChar(20)   // douyin | xiaohongshu | bilibili | kuaishou
  platformUserId    String?  @db.VarChar(200)  // sec_user_id / user_id / uid

  // Pipeline status
  status            String   @default("pending") @db.VarChar(20)
  // pending → scraping → enriching → analyzing → completed → failed
  currentStep       String?  @db.VarChar(20)
  errorMessage      String?  @db.Text

  // Raw data (JSON)
  rawAccountData    Json?    // Platform-returned account info
  rawVideoData      Json?    // Video list + stats
  rawCommentData    Json?    // Comment samples

  // Computed metrics
  metricsData       Json?    // Quantitative metrics (engagement rate, posting frequency, etc.)

  // AI analysis result
  analysisResult    Json?    // Full report JSON (6-dimension scores + narrative)
  overallScore      Int?     // Overall score 0-100

  // Cached display data
  accountName       String?  @db.VarChar(100)
  accountAvatar     String?  @db.VarChar(500)
  followerCount     Int?
  videoCount        Int?

  // Timestamps
  createdAt         DateTime @default(now())
  updatedAt         DateTime @updatedAt
  completedAt       DateTime?

  // Cost tracking
  apiCostUsd        Float?   @default(0)

  @@index([userId, createdAt(sort: Desc)])
  @@index([userId, platform])
}
```

```sql
-- CreateTable
CREATE TABLE `CompetitorAnalysis` (
    `id` VARCHAR(191) NOT NULL,
    `userId` VARCHAR(191) NOT NULL,
    `targetUrl` VARCHAR(500) NOT NULL,
    `platform` VARCHAR(20) NOT NULL,
    `platformUserId` VARCHAR(200) NULL,
    `status` VARCHAR(20) NOT NULL DEFAULT 'pending',
    `currentStep` VARCHAR(20) NULL,
    `errorMessage` TEXT NULL,
    `rawAccountData` JSON NULL,
    `rawVideoData` JSON NULL,
    `rawCommentData` JSON NULL,
    `metricsData` JSON NULL,
    `analysisResult` JSON NULL,
    `overallScore` INTEGER NULL,
    `accountName` VARCHAR(100) NULL,
    `accountAvatar` VARCHAR(500) NULL,
    `followerCount` INTEGER NULL,
    `videoCount` INTEGER NULL,
    `createdAt` DATETIME(3) NOT NULL DEFAULT CURRENT_TIMESTAMP(3),
    `updatedAt` DATETIME(3) NOT NULL,
    `completedAt` DATETIME(3) NULL,
    `apiCostUsd` DOUBLE NULL DEFAULT 0,

    INDEX `CompetitorAnalysis_userId_createdAt_idx`(`userId`, `createdAt` DESC),
    INDEX `CompetitorAnalysis_userId_platform_idx`(`userId`, `platform`),
    PRIMARY KEY (`id`)
) DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;

-- AddForeignKey
ALTER TABLE `CompetitorAnalysis` ADD CONSTRAINT `CompetitorAnalysis_userId_fkey` FOREIGN KEY (`userId`) REFERENCES `User`(`id`) ON DELETE RESTRICT ON UPDATE CASCADE;
```
