    **POLLING LOGIC:**

    ```typescript
    const TERMINAL_STATUSES = new Set(['completed', 'failed'])
    const POLL_INTERVAL = 3000  // 3 seconds

    // In component:
    const params = useParams<{ id: string }>()
    const [analysis, setAnalysis] = useState<ApiCompetitorAnalysis | null>(null)
    const [loading, setLoading] = useState(true)
    const [notFound, setNotFound] = useState(false)

    useEffect(() => {
      async function fetch() {
        try {
          const data = await getCompetitorAnalysis(params.id)
          setAnalysis(data)
        } catch {
          setNotFound(true)
        } finally {
          setLoading(false)
        }
      }
      fetch()
    }, [params.id])

    // Poll until terminal state
    useEffect(() => {
      if (!analysis || TERMINAL_STATUSES.has(analysis.status)) return
      const interval = setInterval(async () => {
        try {
          const data = await getCompetitorAnalysis(params.id)
          setAnalysis(data)
          if (TERMINAL_STATUSES.has(data.status)) clearInterval(interval)
        } catch {
          clearInterval(interval)
        }
      }, POLL_INTERVAL)
      return () => clearInterval(interval)
    }, [analysis?.status, params.id])
    ```

    **POSTING TIME HEATMAP:**

    The heatmap key format from the API is `"Mon-09"`, `"Tue-14"`, etc. (day-hour).

    Render a 24-column × 7-row CSS grid. Days: Sun, Mon, Tue, Wed, Thu, Fri, Sat → labels 日, 一, 二, 三, 四, 五, 六. Hours 0–23 shown as column headers (abbreviated: just the hour number).

    Color intensity per cell: find max value in heatmap, then map `value/max` to opacity classes:
    - 0 → `bg-muted/30`
    - 0.01–0.25 → `bg-primary/20`
    - 0.26–0.5 → `bg-primary/40`
    - 0.51–0.75 → `bg-primary/60`
    - 0.76–1.0 → `bg-primary/80`

    ```tsx
    const days = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat']
    const dayLabels: Record<string, string> = { Sun: '日', Mon: '一', Tue: '二', Wed: '三', Thu: '四', Fri: '五', Sat: '六' }
    const hours = Array.from({ length: 24 }, (_, i) => i)

    // Build lookup: { "Mon-09": 3, ... }
    const heatmap = stats.posting_heatmap

    function cellIntensity(day: string, hour: number, max: number): string {
      const key = `${day}-${String(hour).padStart(2, '0')}`
      const v = heatmap[key] ?? 0
      if (max === 0 || v === 0) return 'bg-muted/30'
      const ratio = v / max
      if (ratio <= 0.25) return 'bg-primary/20'
      if (ratio <= 0.5) return 'bg-primary/40'
      if (ratio <= 0.75) return 'bg-primary/60'
      return 'bg-primary/80'
    }

    const maxVal = Math.max(0, ...Object.values(heatmap))
    ```
