1. Create `apps/web/src/components/marketing/language-switcher.tsx` — Client Component:
   ```typescript
   "use client"

   import { useRouter } from "next/navigation"
   import { Button } from "@/components/ui/button"

   export function LanguageSwitcher({ currentLocale }: { currentLocale: string }) {
     const router = useRouter()

     const toggle = () => {
       const next = currentLocale === "zh" ? "en" : "zh"
       document.cookie = `locale=${next}; path=/; max-age=31536000; SameSite=Lax`
       router.refresh()
     }

     return (
       <Button variant="ghost" size="sm" onClick={toggle} className="cursor-pointer text-sm font-medium">
         {currentLocale === "zh" ? "EN" : "中文"}
       </Button>
     )
   }
   ```
   CRITICAL: Cookie name MUST be `"locale"` — matches i18n/request.ts. max-age=31536000 (1 year). SameSite=Lax. Path=/.
   Uses `router.refresh()` (NOT `window.location.reload()`) to trigger RSC re-render without full page reload.

         : "ClipFlow - AI Video Marketing Pipeline",
       description: isZh
         ? "小企业主的 AI 视频创作助手——输入文案，自动生成专业营销短视频"

       openGraph: {
         title: isZh ? "ClipFlow - AI 营销短视频流水线" : "ClipFlow - AI Video Marketing Pipeline",
         description: isZh
           ? "小企业主的 AI 视频创作助手"
           : "AI-powered video creation for small businesses",
         images: [{ url: "/og-image.png", width: 1200, height: 630 }],
         locale: isZh ? "zh_CN" : "en_US",
         type: "website",
       },
       alternates: {
         canonical: "https://clipflow.ai/",
         languages: {
           "zh-CN": "https://clipflow.ai/",
           "en": "https://clipflow.ai/",
           "x-default": "https://clipflow.ai/",
