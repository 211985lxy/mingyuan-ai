### Task 4: Update ip-profile.ts utility functions (version-gated logic)
**Action**: Edit `apps/web/src/lib/ip-profile.ts`

**4a. Update isIpProfileComplete()**:
```typescript
export function isIpProfileComplete(
  profile: Partial<IpProfile>
): boolean {
  const version = profile.profileVersion ?? 1

  if (version === 1) {
    // Existing logic: check 7 flat fields
    return getIpProfileMissingFields(profile).length === 0
  }

  if (version === 2) {
    // v2: check 3D positioning fields presence
    return !!(
      profile.business &&
      profile.persona &&
      profile.content
    )
  }

  return false
}
```

**4b. Update getIpProfileMissingFields()**:
```typescript
export function getIpProfileMissingFields(
  profile: Partial<IpProfile>
): string[] {
  const version = profile.profileVersion ?? 1

  if (version === 1) {
    // Existing logic
    return REQUIRED_FIELDS.filter((field) => {
      const value = profile[field]
      return typeof value !== "string" || value.trim() === ""
    })
  }

  if (version === 2) {
    const missing: string[] = []
    if (!profile.business) missing.push("business")
    if (!profile.persona) missing.push("persona")
    if (!profile.content) missing.push("content")
    return missing
  }

  return []
}
```

**4c. Update buildIpProfilePromptSnapshot()**:

export function buildIpProfilePromptSnapshot(
  profile: Partial<IpProfile>
): string {
  const version = profile.profileVersion ?? 1

  if (version === 1) {
    // Existing logic (current implementation)
    const displayName = profile.displayName?.trim() || profile.nickname?.trim() || "未命名IP"
    const sections = [
      `你正在为个人 IP「${displayName}」创作文案。`,
      `行业：${profile.industry?.trim() || "未提供"}`,
      `主营内容：${profile.primaryOffer?.trim() || "未提供"}`,
      `目标受众：${profile.targetAudience?.trim() || "未提供"}`,
      `IP特征：${profile.ipTraits?.trim() || "未提供"}`,
      `表达口吻：${profile.toneOfVoice?.trim() || "未提供"}`,
      `可信背书：${profile.proofPoints?.trim() || "未提供"}`,
      `行动号召：${profile.callToAction?.trim() || "未提供"}`,
    ]
    return sections.join("\n")
  }

  if (version === 2 && profile.business && profile.persona && profile.content) {
    // v2: use 3D positioning structure
    const business = profile.business as any
    const persona = profile.persona as any
    const content = profile.content as any

    const displayName = profile.displayName?.trim() || "未命名IP"

    const sections = [
      `你正在为个人 IP「${displayName}」创作文案。`,
      ``,
      `【商业定位】`,
      `核心定位：${business.core || "未提供"}`,
      `目标人群：${business.audience || "未提供"}`,
      `核心价值：${business.value || "未提供"}`,
      `差异化：${business.differentiator || "未提供"}`,
      ``,
      `【人设设计】`,
      `专业度：${persona.expertiseLevel || "未提供"}`,
      `表达风格：${persona.expressionStyle || "未提供"}`,
      `人设标签：${persona.traits?.join("、") || "未提供"}`,
      ``,
      `【内容策略】`,
      `内容主题：${content.themes?.map((t: any) => `${t.name}(${t.ratio}%)`).join("、") || "未提供"}`,
      `内容形式：${content.formats?.join("、") || "未提供"}`,
      `更新节奏：${content.rhythm || "未提供"}`,
    ]
    return sections.join("\n")
  }

  // Fallback for incomplete v2
  return `个人 IP 档案未完成，无法生成提示词快照。`
}

**Action**: Edit `apps/web/src/app/api/ip-profile/route.ts`

**In PUT handler**:
```typescript
export const PUT = withUserAuth(async (request, { user }) => {
  const body = await request.json()

  // Determine version from request body
  const version = body.profileVersion ?? 1

  let updateData: any
  let isComplete: boolean
  let promptSnapshot: string | null = null

  if (version === 1) {
    // v1 update: use existing logic
    const normalized = normalizeIpProfileInput(body)
    isComplete = isIpProfileComplete(normalized)
    promptSnapshot = buildIpProfilePromptSnapshot(normalized)

    updateData = {
      ...normalized,
      profileVersion: 1,
      isComplete,
      promptSnapshot,
      isActive: true,
    }
  } else if (version === 2) {
    // v2 update: accept survey + positioning fields
    const profile = {
      profileVersion: 2,
      displayName: body.displayName,
      nickname: body.nickname,
      surveyIndustry: body.surveyIndustry,
      surveyTargetCustomer: body.surveyTargetCustomer,
      surveyMonetization: body.surveyMonetization,
      surveyPersonalTraits: body.surveyPersonalTraits,
      surveyContentGoal: body.surveyContentGoal,
      business: body.business,
      persona: body.persona,
      content: body.content,
    }

    isComplete = isIpProfileComplete(profile)
    promptSnapshot = isComplete ? buildIpProfilePromptSnapshot(profile) : null

    updateData = {
      ...profile,
      isComplete,
      promptSnapshot,
      isActive: true,
    }
  } else {
    return NextResponse.json(
      { error: "Invalid profileVersion" },
      { status: 400 }
    )
  }

  const profile = await prisma.ipProfile.upsert({
    where: { userId: user.id },
    update: updateData,
    create: {
      userId: user.id,
      ...updateData,
    },
  })

  const missingFields = getIpProfileMissingFields(profile)

  return NextResponse.json({
    data: {
      profile,
      isComplete,
      missingFields,
      promptSnapshot,
    },
  })
})
```

**Verify**:

- PUT with profileVersion=2 and v2 fields saves correctly
- isComplete correctly calculated for v1 vs v2
- promptSnapshot generated correctly for v1 vs v2
