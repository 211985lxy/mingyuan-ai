<objective>
Create and deploy an OSS lifecycle policy that transitions video files from Standard storage to Infrequent Access (IA) tier, reducing storage costs by approximately 50% for older videos.

Purpose: 4K videos are 3-4x larger than 1080p. Without lifecycle management, storage costs will grow linearly with video count. IA transition after the initial access window (7 days for 1080p, 30 days for 4K) reduces costs without affecting video availability.

Output: Lifecycle policy JSON file, programmatic deployment utility, and verified policy on production bucket.
</objective>

2. **Create `infra/oss-lifecycle-policy.json`** with two lifecycle rules:

Rule 1 — `transition-1080p-to-ia`:
- Prefix: `videos/`
- Status: Enabled
- Transition: Standard to IA after 7 days
- Filter: ObjectSizeGreaterThan 1MB (1048576 bytes), ObjectSizeLessThan 100MB (104857600 bytes)
- This catches 1080p videos (typically 5-50MB) but not tiny files (thumbnails, covers)
- Note: Actual ObjectSizeGreaterThan/LessThan are standard Aliyun OSS lifecycle filter fields

Rule 2 — `transition-4k-to-ia`:
- Prefix: `videos/`
- Status: Enabled
- Transition: Standard to IA after 30 days
- Filter: ObjectSizeGreaterThan 100MB (104857600 bytes)
- This catches 4K videos (typically 100-400MB), gives 30 days of standard-tier access since 4K is the premium feature users access more initially

Both rules:
- No expiration/deletion (user data preservation)
- No AbortMultipartUpload (not relevant)

3. **Create `apps/web/src/lib/oss-lifecycle.ts`** as a utility for programmatic lifecycle management:

```typescript
import OSS from "ali-oss";

/**
 * Apply lifecycle rules to the configured OSS bucket.
 *
 * Usage: npx tsx src/lib/oss-lifecycle.ts
 *
 * Transitions videos/ prefix objects to Infrequent Access (IA) tier:
 * - 1080p videos (1MB-100MB): Standard -> IA after 7 days
 * - 4K videos (>100MB): Standard -> IA after 30 days
 *
 * IA storage is ~50% cheaper than Standard for Aliyun OSS.
 * Objects remain fully accessible (IA has retrieval fee but no access delay).
 */

const LIFECYCLE_RULES: OSS.LifecycleRule[] = [
  {
    id: "transition-1080p-to-ia",
    prefix: "videos/",
    status: "Enabled" as const,
    transitions: [
      {
        days: 7,
        storageClass: "IA" as const,
      },
    ],
    filter: {
      objectSizeGreaterThan: 1048576,      // 1 MB
      objectSizeLessThan: 104857600,       // 100 MB
    },
  },
  {
    id: "transition-4k-to-ia",
    prefix: "videos/",
    status: "Enabled" as const,
    transitions: [
      {
        days: 30,
        storageClass: "IA" as const,
      },
    ],
    filter: {
      objectSizeGreaterThan: 104857600,    // 100 MB
    },
  },
];

function createOssClient(): OSS {
  const region = process.env.ALIBABA_CLOUD_OSS_REGION;
  const bucket = process.env.ALIBABA_CLOUD_OSS_BUCKET;
  const accessKeyId = process.env.ALIBABA_CLOUD_ACCESS_KEY_ID;
  const accessKeySecret = process.env.ALIBABA_CLOUD_ACCESS_KEY_SECRET;

  if (!region || !bucket || !accessKeyId || !accessKeySecret) {
    throw new Error(
      "Missing OSS config. Required env vars: ALIBABA_CLOUD_OSS_REGION, ALIBABA_CLOUD_OSS_BUCKET, ALIBABA_CLOUD_ACCESS_KEY_ID, ALIBABA_CLOUD_ACCESS_KEY_SECRET"
    );
  }

  return new OSS({ region, bucket, accessKeyId, accessKeySecret });
}

export async function applyLifecyclePolicy(): Promise<void> {
  const client = createOssClient();
  const bucket = process.env.ALIBABA_CLOUD_OSS_BUCKET!;

  console.log(`[oss-lifecycle] Applying lifecycle rules to bucket: ${bucket}`);
  console.log(`[oss-lifecycle] Rules: ${LIFECYCLE_RULES.map(r => r.id).join(", ")}`);

  await client.putBucketLifecycle(bucket, LIFECYCLE_RULES);

  console.log("[oss-lifecycle] Lifecycle rules applied successfully.");
  console.log("[oss-lifecycle] Verifying...");

  const result = await client.getBucketLifecycle(bucket);
  const rules = result.rules || [];
  console.log(`[oss-lifecycle] Active rules: ${rules.length}`);
  for (const rule of rules) {
    console.log(`  - ${rule.id}: prefix=${rule.prefix}, status=${rule.status}`);
  }
}

export async function getLifecyclePolicy(): Promise<OSS.LifecycleRule[]> {
  const client = createOssClient();
  const bucket = process.env.ALIBABA_CLOUD_OSS_BUCKET!;
  const result = await client.getBucketLifecycle(bucket);
  return result.rules || [];
}

// CLI entry point
if (require.main === module || process.argv[1]?.endsWith("oss-lifecycle.ts")) {
  applyLifecyclePolicy()
    .then(() => {
      console.log("[oss-lifecycle] Done.");
      process.exit(0);
    })
    .catch((err) => {
      console.error("[oss-lifecycle] Failed:", err.message);
      process.exit(1);
    });
}
```
